# Duplicate address again to retrieve and check the char
 \n                            -> DUP
\t\t\t                         -> RETRIEVE

# Push 10 (newline ASCII) to compare
   \t \t \n                    -> PUSH 10

# Subtract to see if char == newline
\t  \t                         -> SUB

# If char - 10 == 0, jump to END_READ
\n\t \t\n                      -> JUMP_ZERO END_READ

# Else: increment target heap address by 1
   \t\n                        -> PUSH 1
\t                             -> ADD

# Loop back to READ_LOOP
\n \n \n                       -> JUMP READ_LOOP

# Mark label END_READ
\n  \t\n                       -> LABEL END_READ

# Overwrite the newline character with a 0 sentinel
